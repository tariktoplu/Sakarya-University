ORG 0000H
SJMP BASLA
ORG 0003H
	LJMP KESME_INT0
ORG 000BH 
	LJMP KESME_TR0
ORG 0013H
	LJMP KESME_INT1

ORG 0030H
	BASLA:
SETB IT0 ;DUSEN KENAR TETIKLEME
SETB IT1 ;DUSEN KENAR TETIKLEME
MOV IE,#87H ;EA=1 EX0,1=1 ET0=1

SETB PX1 ;SENSON ONCELIGI
MOV TMOD,#12H ;TIMER1MOD=1(65536) TIMER2MOD=2 (256)
SETB P1.1 ;KIRMIZI LAMBA YAK
CLR P1.2 ;YESIL LAMBA SONDUR
CLR P1.0 ;KOL INIK SEVIYEDE
MOV TL0,#-250 ;BASLANGIC DEGERI
MOV TH0,#-250 ;OTOMATIK YUKLEME



KESME_INT0:
SETB P1.0 ;MOTORU CALISTIR
CLR P1.1 ;KIRMIZI LAMBA SONDUR
SETB TR0 ;YESIL LAMBA ICIN TIMER CALISTIR

CALL ZAMAN_BEKLE
SETB P1.1 ;KIRMIZI LAMBA YAK
CLR P1.2 ;YESIL LAMBA SONDUR
CLR P1.0 ;MOTORU KAPAT(BARIYERI INDIR)
RETI

KESME_INT1: ;ENGEL GELDIGI DURUM
SETB P1.0 ;MOTORU CALISTIR
SETB TR0 ;YESIL LAMBA ICIN TIMER CALISTIR
CLR P1.1 ;KIRMIZI LAMBA SONDUR

CALL ZAMAN_BEKLE
RETI

KESME_TR0:
CPL P1.2
RETI

ZAMAN_BEKLE:
MOV TH1,#HIGH(-50000)
MOV TL1,#LOW(-50000)
SETB TR1
BEKLE:JNB TF1,BEKLE
RET
END
